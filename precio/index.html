// 🔥 DIBUJAR LAS TARJETAS DE GRUPOS DE COMPATIBILIDAD function renderGruposCompatibles(grupos, term) { container.innerHTML = ''; resultsCount.innerHTML = `${grupos.length} grupo(s) encontrado(s) Modo Catálogo`; if (grupos.length === 0) { container.innerHTML = `

Sin resultados

No hay compatibles registrados para "${escapeHtml(term)}"

`; return; } const view = document.createElement('div'); view.className = 'cards-view'; grupos.forEach(g => { // 🔥 AQUÍ LEEMOS DIRECTAMENTE LO QUE DICE EL DICCIONARIO const listaHtml = g.modelos.map(m => { const partes = m.trim().split(" "); const marca = partes[0]; // Primera palabra const modelo = partes.slice(1).join(" "); // Todo el resto return `
${escapeHtml(marca)} ${escapeHtml(modelo)}
`; }).join(''); const card = document.createElement('div'); card.className = 'product-card is-compat'; card.innerHTML = `
GRUPO DE COMPATIBILIDAD
${escapeHtml(g.marca)} ${escapeHtml(g.grupoNombre)}
Modelos Compatibles (${g.modelos.length}):
${listaHtml}
`; view.appendChild(card); }); container.appendChild(view); renderedCount = 1; }
