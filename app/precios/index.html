// ========================================== // 🔥 LÓGICA DEL MODO TÉCNICO CON CONFIRMACIÓN // ========================================== let modoTecnicoActivo = false; // Al cargar, verificamos si ya estaba activo en localStorage if (localStorage.getItem('drx_modo_tecnico') === 'true') { modoTecnicoActivo = true; document.getElementById('techModeBtn')?.classList.add('tech-active'); } document.getElementById('techModeBtn')?.addEventListener('click', () => { if (modoTecnicoActivo) { // Si ya está activo, lo desactivamos directamente modoTecnicoActivo = false; localStorage.setItem('drx_modo_tecnico', 'false'); document.getElementById('techModeBtn')?.classList.remove('tech-active'); toast('Modo Técnico desactivado', 'info'); renderizarProductos(); // Vuelve a renderizar sin precios de compra } else { // Si no está activo, mostramos el modal de confirmación document.getElementById('modalTechConfirm').classList.add('show'); } }); document.getElementById('btnCancelTech')?.addEventListener('click', () => { document.getElementById('modalTechConfirm').classList.remove('show'); }); document.getElementById('btnConfirmTech')?.addEventListener('click', () => { modoTecnicoActivo = true; localStorage.setItem('drx_modo_tecnico', 'true'); document.getElementById('techModeBtn')?.classList.add('tech-active'); document.getElementById('modalTechConfirm').classList.remove('show'); toast('Modo Técnico activado. Mostrando precios de compra.', 'success'); renderizarProductos(); // Vuelve a renderizar con precios de compra });
